    if (name === "brand") field = document.getElementById("ymwasher-brand");
    else if (name === "fault") field = document.getElementById("ymwasher-fault-type");
    else if (name === "urgency") field = document.getElementById("ymwasher-urgency");
    else if (name === "consent") field = document.getElementById("ymwasher-consent-check");
    if (errEl) { if (show) errEl.classList.add("ymwasher-show"); else errEl.classList.remove("ymwasher-show"); }
    if (field && field.classList) { if (show) field.classList.add("ymwasher-error"); else field.classList.remove("ymwasher-error"); }
  }


  function validateForm() {
    var valid = true;
    var brand = document.getElementById("ymwasher-brand");
    var fault = document.getElementById("ymwasher-fault-type");
    var urgency = document.getElementById("ymwasher-urgency");
    var consent = document.getElementById("ymwasher-consent-check");
    if (!brand || !brand.value) { showError("brand", true); valid = false; } else showError("brand", false);
    if (!fault || !fault.value) { showError("fault", true); valid = false; } else showError("fault", false);
    if (!urgency || !urgency.value) { showError("urgency", true); valid = false; } else showError("urgency", false);
    if (!consent || !consent.checked) { showError("consent", true); valid = false; } else showError("consent", false);
    return valid;
  }


  function handleCalcSubmit(e) {
    e.preventDefault();
    if (!validateForm()) { trackGoal("form_validation_error"); showToast("Проверьте обязательные поля."); return; }
    var brand = document.getElementById("ymwasher-brand").value;
    var fault = document.getElementById("ymwasher-fault-type").value;
    var urgency = document.getElementById("ymwasher-urgency").value;
    var comment = document.getElementById("ymwasher-comment").value;
    var message = buildCalcMessage(brand, fault, urgency, comment);
    trackGoal("form_submit", { brand: brand, fault: fault, urgency: urgency });
    openWhatsApp(message);
  }


  function captureUTM() {
    var params = {};
    var search = window.location.search;
    if (!search) return params;
    var pairs = search.substring(1).split("&");
    for (var i = 0; i < pairs.length; i++) {
      var pair = pairs[i].split("=");
      var key = decodeURIComponent(pair[0]).toLowerCase();
      var val = pair[1] ? decodeURIComponent(pair[1]) : "";
      if (key.indexOf("utm_") === 0) params[key] = val;
    }
    return params;
  }


  function storeUTM() {
    try {
      var utm = captureUTM();
      var hasUtm = false;
      for (var k in utm) { if (utm.hasOwnProperty(k)) { hasUtm = true; break; } }
      if (hasUtm) sessionStorage.setItem("ymwasher_utm", JSON.stringify(utm));
    } catch (e) {}
  }


  function initRevealAnimations() {
    if (!("IntersectionObserver" in window)) {
      var els = document.querySelectorAll(".ymwasher-reveal");
      for (var i = 0; i < els.length; i++) els[i].classList.add("ymwasher-visible");
      return;
    }
    var observer = new IntersectionObserver(function (entries) {
      for (var i = 0; i < entries.length; i++) {
        if (entries[i].isIntersecting) { entries[i].target.classList.add("ymwasher-visible"); observer.unobserve(entries[i].target); }
      }
    }, { threshold: 0.12, rootMargin: "0px 0px -60px 0px" });
    var revealEls = document.querySelectorAll(".ymwasher-reveal");
    for (var j = 0; j < revealEls.length; j++) observer.observe(revealEls[j]);
  }


  function openPolicy(e) { e.preventDefault(); showToast("Политика конфиденциальности будет добавлена отдельно."); }


  function init() {
    applyPhone();
    storeUTM();
    var yearEl = document.querySelector("[data-ymwasher-year]");
    if (yearEl) yearEl.textContent = new Date().getFullYear();


    var scrollLinks = document.querySelectorAll("[data-ymwasher-scroll]");
    for (var i = 0; i < scrollLinks.length; i++) {
      scrollLinks[i].addEventListener("click", function (e) {
        var href = this.getAttribute("href") || "";
        if (href.charAt(0) === "#" && href.length > 1) { e.preventDefault(); smoothScrollTo(href.substring(1)); closeMobileMenu(); }
      });
    }


    var actionBtns = document.querySelectorAll("[data-ymwasher-action]");
    for (var j = 0; j < actionBtns.length; j++) {
      actionBtns[j].addEventListener("click", function (e) {
        e.preventDefault();
        handleAction(this.getAttribute("data-ymwasher-action"));
        if (this.hasAttribute("data-ymwasher-close-menu")) closeMobileMenu();
      });
    }


    var trackEls = document.querySelectorAll("[data-ymwasher-track]");
    for (var t = 0; t < trackEls.length; t++) {
      trackEls[t].addEventListener("click", function () { var goal = this.getAttribute("data-ymwasher-track"); if (goal) trackGoal(goal); });
    }


    var burger = document.querySelector(".ymwasher-burger");
    if (burger) burger.addEventListener("click", function () { if (this.classList.contains("ymwasher-active")) closeMobileMenu(); else openMobileMenu(); });
    var overlay = document.querySelector(".ymwasher-mobile-overlay");
    if (overlay) overlay.addEventListener("click", closeMobileMenu);


    var closeMenuLinks = document.querySelectorAll("[data-ymwasher-close-menu]");
    for (var c = 0; c < closeMenuLinks.length; c++) closeMenuLinks[c].addEventListener("click", closeMobileMenu);


    document.addEventListener("keydown", function (e) { if (e.key === "Escape") closeMobileMenu(); });


    var faultCards = document.querySelectorAll(".ymwasher-fault-card");
    for (var f = 0; f < faultCards.length; f++) {
      faultCards[f].addEventListener("click", function () { selectFault(this.getAttribute("data-ymwasher-fault")); });
    }


    var faqQuestions = document.querySelectorAll(".ymwasher-faq-question");
    for (var q = 0; q < faqQuestions.length; q++) {
      faqQuestions[q].addEventListener("click", function () { var item = this.closest(".ymwasher-faq-item"); if (item) toggleFaq(item); });
    }


    var form = document.getElementById("ymwasher-calc-form");
    if (form) form.addEventListener("submit", handleCalcSubmit);


    var brandField = document.getElementById("ymwasher-brand");
    if (brandField) brandField.addEventListener("change", function () { showError("brand", false); });
    var faultField = document.getElementById("ymwasher-fault-type");
    if (faultField) faultField.addEventListener("change", function () { showError("fault", false); });
    var urgencyField = document.getElementById("ymwasher-urgency");
    if (urgencyField) urgencyField.addEventListener("change", function () { showError("urgency", false); });
    var consentCheck = document.getElementById("ymwasher-consent-check");
    if (consentCheck) consentCheck.addEventListener("change", function () { showError("consent", false); });


    var policyLinks = document.querySelectorAll("[data-ymwasher-open-policy]");
    for (var p = 0; p < policyLinks.length; p++) policyLinks[p].addEventListener("click", openPolicy);


    initRevealAnimations();
    trackGoal("page_view");
  }


  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", init);
  else init();
})();
